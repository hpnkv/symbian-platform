cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(central_repository_app_classic LANGUAGES CXX ASM)
endif()
include(SymbianApp)
symbian_add_executable(central_repository_app_classic main.cc ui.cc)
target_link_libraries(central_repository_app_classic PRIVATE Symbian::WindowServer Symbian::Gdi)
target_link_libraries(central_repository_app_classic PRIVATE
  Symbian::CentralRepository)
symbian_enable_native_leaves(central_repository_app_classic SOURCES main.cc)
symbian_publish_executable(central_repository_app_classic UID3 0xe0000e0d)
