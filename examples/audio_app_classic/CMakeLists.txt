cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(audio_app_classic LANGUAGES CXX ASM)
endif()
include(SymbianApp)
symbian_add_executable(audio_app_classic app.cc)
target_link_libraries(audio_app_classic PRIVATE Symbian::AudioStream)
symbian_publish_executable(audio_app_classic UID3 0xe0000e03)
symbian_enable_native_leaves(audio_app_classic SOURCES app.cc)
