cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(vibra_app_classic LANGUAGES CXX ASM)
endif()
include(SymbianApp)
symbian_add_executable(vibra_app_classic app.cc ui.cc)
target_link_libraries(vibra_app_classic PRIVATE Symbian::WindowServer Symbian::Gdi)
target_link_libraries(vibra_app_classic PRIVATE Symbian::Vibra)
symbian_publish_executable(vibra_app_classic UID3 0xe0000e01)
symbian_enable_native_leaves(vibra_app_classic SOURCES app.cc)
