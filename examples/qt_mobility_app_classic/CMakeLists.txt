cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(qt_mobility_app_classic LANGUAGES CXX ASM)
endif()
include(SymbianApp)
symbian_add_executable(qt_mobility_app_classic main.cc ui.cc)
target_link_libraries(qt_mobility_app_classic PRIVATE Symbian::WindowServer Symbian::Gdi)
target_link_libraries(qt_mobility_app_classic PRIVATE
  Symbian::QtMobilityContacts Symbian::QtMobilityLocation)
symbian_publish_executable(qt_mobility_app_classic UID3 0xe0000e0a)
