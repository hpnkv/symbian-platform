cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(openc_app_classic LANGUAGES CXX ASM)
endif()

target_link_libraries(openc_app_classic PRIVATE Symbian::OpenC)
symbian_publish_executable(openc_app_classic UID3 0xe0000e0b)
