cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(image_app_classic LANGUAGES CXX ASM)
endif()
include(SymbianApp)
symbian_add_executable(image_app_classic app.cc ui.cc)
target_link_libraries(image_app_classic PRIVATE Symbian::WindowServer Symbian::Gdi)
target_link_libraries(image_app_classic PRIVATE Symbian::ImageConversion)
symbian_publish_executable(image_app_classic UID3 0xe0000e04)
symbian_enable_native_leaves(image_app_classic SOURCES app.cc)
