cmake_minimum_required(VERSION 3.28)
project(symbian_device_api LANGUAGES C CXX)

if(NOT SYMBIAN_SDK_PREFIX)
  message(FATAL_ERROR "Device API build requires SYMBIAN_SDK_PREFIX")
endif()
if(NOT TARGET Symbian::Runtime)
  include("${SYMBIAN_SDK_PREFIX}/cmake/SymbianApp.cmake")
endif()
if(NOT TARGET Symbian::AbseilStatusOr)
  message(FATAL_ERROR "Device APIs require the installed Abseil StatusOr profile")
endif()

# Each implemented capability owns its archive and native service boundary.
add_subdirectory(system)
add_subdirectory(../net net)
add_subdirectory(../http http)
add_subdirectory(../websocket websocket)
add_subdirectory(connectivity)
add_subdirectory(storage)
add_subdirectory(agent)
add_subdirectory(power)
add_subdirectory(display)
add_subdirectory(media)
add_subdirectory(camera)

get_filename_component(canary_root "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${canary_root}/cmake/SymbianGuestHeaderCanaries.cmake")
symbian_guest_api_header_canaries("${canary_root}")
